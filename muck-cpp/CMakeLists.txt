cmake_minimum_required(VERSION 3.16)
project(MuckCpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Debug)
endif()
add_executable(muck src/main.cpp src/game.cpp)
if(MSVC)
  target_compile_options(muck PRIVATE /W4)
else()
  target_compile_options(muck PRIVATE -Wall -Wextra)
endif()
